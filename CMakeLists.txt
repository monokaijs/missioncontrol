cmake_minimum_required(VERSION 3.24)
project(MissionControl VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MC_VERSION "${PROJECT_VERSION}" CACHE STRING "Release version")
if(NOT MC_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
 message(FATAL_ERROR "MC_VERSION must be numeric major.minor.patch")
endif()
# nlohmann/json 3.12.0 (MIT), immutable commit and independently verified SHA-256.
set(JSON_INCLUDE "${CMAKE_BINARY_DIR}/_deps/include")
file(MAKE_DIRECTORY "${JSON_INCLUDE}/nlohmann")
if(NOT EXISTS "${JSON_INCLUDE}/nlohmann/json.hpp")
 file(DOWNLOAD "https://raw.githubusercontent.com/nlohmann/json/65ee68451d8eb2b5f3a30b410476ab83deb3289b/single_include/nlohmann/json.hpp"
  "${JSON_INCLUDE}/nlohmann/json.hpp" EXPECTED_HASH SHA256=aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63 TLS_VERIFY ON)
endif()
file(SHA256 "${JSON_INCLUDE}/nlohmann/json.hpp" JSON_HASH)
if(NOT JSON_HASH STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
 message(FATAL_ERROR "JSON dependency integrity failure")
endif()
add_library(mc_policy INTERFACE)
target_include_directories(mc_policy INTERFACE src "${JSON_INCLUDE}")
if(MSVC)
 add_compile_options(/W4 /permissive- /utf-8 /EHsc /guard:cf)
 add_link_options(/DYNAMICBASE /NXCOMPAT /guard:cf)
endif()
include(CTest)
add_executable(policy_tests tests/policy_tests.cpp)
target_link_libraries(policy_tests PRIVATE mc_policy)
add_test(NAME security_policy COMMAND policy_tests)
if(WIN32)
 add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
 string(REPLACE "." "," MC_VERSION_COMMA "${MC_VERSION}")
 configure_file(src/app.manifest.in "${CMAKE_BINARY_DIR}/app.manifest" @ONLY)
 configure_file(src/app.rc.in "${CMAKE_BINARY_DIR}/app.rc" @ONLY)
 add_executable(MissionControl WIN32 src/main.cpp src/platform.cpp src/telegram.cpp "${CMAKE_BINARY_DIR}/app.rc")
 target_link_options(MissionControl PRIVATE /MANIFEST:NO)
 target_link_libraries(MissionControl PRIVATE mc_policy winhttp crypt32 bcrypt windowscodecs ole32 shell32 shlwapi advapi32 user32 gdi32 comctl32 uuid)
 target_compile_definitions(MissionControl PRIVATE MC_VERSION="${MC_VERSION}")
 add_executable(windows_tests tests/windows_tests.cpp src/platform.cpp)
 target_link_libraries(windows_tests PRIVATE mc_policy crypt32 bcrypt windowscodecs ole32 shell32 shlwapi advapi32 user32 gdi32 uuid)
 add_test(NAME windows_safety COMMAND windows_tests)
 add_test(NAME offline_smoke COMMAND MissionControl --self-test)
endif()
